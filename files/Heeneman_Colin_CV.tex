% ---------- Selected Projects & Research ----------
\section*{Selected Projects \& Research}

\cvtitle{\href{https://colin-heeneman.github.io/projects/medicaid-expansion.html}{Medicaid Expansion \& Health Insurance Coverage}}
\cvdetail{Staggered difference-in-differences (Callaway-Sant\textquoteright Anna) analysis of ACA Medicaid expansion using ACS and BRFSS panel data (2010\,--\,2019). Found a 2.3 percentage point reduction in the uninsured rate among non-elderly adults (95\% CI: $-3.5$ to $-1.1$ pp), with a 5.1 pp reduction among adults below 200\% FPL. Includes an accompanying policy brief for state-level decision-makers.}

\cvtitle{\href{https://colin-heeneman.github.io/projects/ada-parc.html}{ADA-PARC Project}}
\cvdetail{Descriptive and index-based analysis of quality-of-life disparities between disabled and non-disabled populations using 2023 ACS 5-year data across all 50 states and hundreds of geographies. Delivered via written report, an R Shiny application, and scorecard dashboards built natively in R and Quarto.}

\cvtitle{\href{https://colin-heeneman.github.io/projects/wic.html}{WIC Program Evaluation}}
\cvdetail{Regression discontinuity design estimating the effect of WIC income-eligibility (185\% FPL) on infant birth weight, using synthetic data to demonstrate the evaluation framework under realistic data constraints. With Emily Frost and Samantha Sepulveda, UIC PA 528.}
